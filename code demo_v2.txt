; ===== UART Temperature Monitor - Optimized =====
.model small
.stack 100h

.data
    ; 8255A addresses
    PORTA   EQU 20h
    PORTB   EQU 21h
    PORTC   EQU 22h
    CTRL    EQU 23h

    ; 8251A addresses
    UART_CMD  EQU 25h
    UART_DATA EQU 24h

    msg1    db 'Temp: $'
    msg2    db 'C, $'
    alarm   db 'ALARM!',13,10,'$'
    okmsg   db 'OK',13,10,'$'
    threshold db 80        ; Threshold for alarm (x2)
    temp    db ?

.code
main proc
    mov ax, @data
    mov ds, ax

    ; --- 8255A setup ---
    mov dx, CTRL
    mov al, 91h
    out dx, al

    ; --- UART 8251A setup ---
    mov dx, UART_CMD
    mov al, 40h
    out dx, al
    mov al, 4Eh
    out dx, al
    mov al, 05h
    out dx, al

main_loop:
    ; ---- ADC0808 Reading ----
    mov dx, PORTB
    xor al, al
    out dx, al             ; Select IN0, START=0
    mov al, 08h
    out dx, al             ; START/ALE high
    xor al, al
    out dx, al             ; START/ALE low

    mov dx, PORTC
wait_eoc:
    in al, dx
    test al, 1
    jnz wait_eoc           ; Wait for EOC (PC0 == 0)

    ; ----- Read converted value -----
    mov dx, PORTA
    in al, dx
    shl al, 1              ; T = D * 2
    mov [temp], al

    ; ---- Print Temp:XXC ----
    mov si, offset msg1
    call print_str

    mov al, [temp]
    mov ah, 0
    mov bl, 10
    div bl                 ; AL = tens, AH = units

    add al, '0'
    call tx_char
    mov al, ah
    add al, '0'
    call tx_char

    mov si, offset msg2
    call print_str

    ; ---- Alarm check and LED control ----
    mov al, [temp]
    cmp al, [threshold]
    ja alarm_on

; ---- Normal Case (<= threshold) ----
    mov dx, PORTC
    in al, dx
    and al, 0EFh           ; Clear bit 4 for OK
    out dx, al

    mov si, offset okmsg
    call print_str
    jmp do_delay

alarm_on:
    mov dx, PORTC
    in al, dx
    or al, 10h             ; Set bit 4 for ALARM
    out dx, al

    mov si, offset alarm
    call print_str

do_delay:
    mov cx, 0FFFFh
delay_loop:
    loop delay_loop
    jmp main_loop

; ---- Print string pointed to by SI (ends at '$') ----
print_str proc
    push ax
print_str_loop:
    lodsb
    cmp al, '$'
    je print_str_end
    call tx_char
    jmp print_str_loop
print_str_end:
    pop ax
    ret
print_str endp

; ---- Send byte in AL via UART ----
tx_char proc
    push ax
    mov dx, UART_CMD
.wait:
    in al, dx
    test al, 01h
    jz .wait
    pop ax
    mov dx, UART_DATA
    out dx, al
    ret
tx_char endp
end main
